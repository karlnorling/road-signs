The `@road-signs/bt` package contains Bhutan road signs.

## Installation

```sh
npm install @road-signs/bt
```

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/bt';

    // Placeholders — replace with real codes once @road-signs/bt ships a registry.

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("example-sign");

    // By sign code
    const s = getSignByCode("EXAMPLE-CODE");

    // All warning signs
    const filtered = getSignsByCategory("warning");
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/bt';
    import { RoadSign } from '@road-signs/react';

    const warningSigns = getSignsByCategory("warning");

    export default function BTGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {warningSigns.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
  <TabItem label="Vue 3">
    ```vue
    <script setup>
    import { getSignsByCategory } from '@road-signs/bt';
    import { RoadSign } from '@road-signs/vue';

    const warningSigns = getSignsByCategory("warning");
    </script>

    <template>
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        <RoadSign v-for="sign in warningSigns" :key="sign.id" :sign="sign" :size="64" />
      </div>
    </template>
    ```
  </TabItem>
</Tabs>
