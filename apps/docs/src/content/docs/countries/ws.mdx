The `@road-signs/ws` package contains Samoa road signs.

## Installation

```sh
npm install @road-signs/ws
```

{/* USAGE:START */}

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/ws';

    // Example below uses the real WS sign "FORD" — adjust to taste.

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("ford-ford");

    // By sign code
    const s = getSignByCode("FORD");

    // All warning signs
    const filtered = getSignsByCategory("warning");
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/ws';
    import { RoadSign } from '@road-signs/react';

    const warningSigns = getSignsByCategory("warning");

    export default function WSGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {warningSigns.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
  <TabItem label="Vue 3">
    ```vue
    <script setup>
    import { getSignsByCategory } from '@road-signs/ws';
    import { RoadSign } from '@road-signs/vue';

    const warningSigns = getSignsByCategory("warning");
    </script>

    <template>
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        <RoadSign v-for="sign in warningSigns" :key="sign.id" :sign="sign" :size="64" />
      </div>
    </template>
    ```
  </TabItem>
</Tabs>
