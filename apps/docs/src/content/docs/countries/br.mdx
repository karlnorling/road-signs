{/* QUICK_FACTS:END */}

The `@road-signs/br` package contains Brazil road signs from the **Código de Trânsito Brasileiro (CTB)**, Brazil's national traffic code.
Signs are sourced from Wikipedia / Wikimedia Commons. Signs are in the public domain.

## Installation

```sh
npm install @road-signs/br
```

## Sign categories

| Category | Series | Description |
| -------- | ------ | ----------- |
| `warning` | A-xx | Advertência — diamond hazard warnings |
| `priority` | — | Give way / stop |
| `prohibitory` | R-xx | Regulamentação — regulatory signs |
| `mandatory` | — | Compulsory actions |
| `information` | D-xx / S-xx | Indicação / Serviço signs |

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/br';

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("a-1-curves-ahead");

    // By sign code
    const s = getSignByCode("A-1");

    // All prohibitory signs
    const filtered = getSignsByCategory('prohibitory');
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/br';
    import { RoadSign } from '@road-signs/react';

    const warnings = getSignsByCategory('warning');

    export default function WarningGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {warnings.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
  <TabItem label="Vue 3">
    ```vue
    <script setup>
    import { getSignsByCategory } from '@road-signs/br';
    import { RoadSign } from '@road-signs/vue';

    const warnings = getSignsByCategory('warning');
    </script>

    <template>
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        <RoadSign v-for="sign in warnings" :key="sign.id" :sign="sign" :size="64" />
      </div>
    </template>
    ```
  </TabItem>

## TypeScript types

```ts
import type { BRSign, BRCategory } from '@road-signs/br';

type BRCategory =
  | 'information'
  | 'mandatory'
  | 'priority'
  | 'prohibitory'
  | 'warning';

interface BRSign {
  id: string;          // e.g. "a-1-curves-ahead"
  code: string;        // e.g. "A-1"
  name: string;
  description: string;
  category: BRCategory;
  svg: string;         // inline SVG string
  assets: {
    svg: string;
    png: Record<number, string>;
    jpg: Record<number, string>;
    webp: Record<number, string>;
  };
}
```
