{/* QUICK_FACTS:END */}

The `@road-signs/pt` package contains Portugal road signs from the **Código da Estrada**, Portugal's road traffic code which follows the Vienna Convention.
Signs are sourced from Wikipedia / Wikimedia Commons. Signs are in the public domain.

## Installation

```sh
npm install @road-signs/pt
```

## Sign categories

| Category | Series | Description |
| -------- | ------ | ----------- |
| `warning` | A series | Sinais de perigo — hazard warnings |
| `priority` | B series | Sinais de cedência de passagem — right-of-way |
| `prohibitory` | C series | Sinais de proibição — prohibitions |
| `mandatory` | D series | Sinais de obrigação — compulsory actions |
| `information` | E–L series | Sinais de informação — information signs |

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/pt';

    // Example below uses the real PT sign "dangerous-curve-to-right" — adjust to taste.

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("dangerous-curve-to-right-dangerous-curve-to-right");

    // By sign code
    const s = getSignByCode("dangerous-curve-to-right");

    // All warning signs
    const filtered = getSignsByCategory("warning");
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/pt';
    import { RoadSign } from '@road-signs/react';

    const warningSigns = getSignsByCategory("warning");

    export default function PTGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {warningSigns.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
  <TabItem label="Vue 3">
    ```vue
    <script setup>
    import { getSignsByCategory } from '@road-signs/pt';
    import { RoadSign } from '@road-signs/vue';

    const warningSigns = getSignsByCategory("warning");
    </script>

    <template>
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        <RoadSign v-for="sign in warningSigns" :key="sign.id" :sign="sign" :size="64" />
      </div>
    </template>
    ```
  </TabItem>

## TypeScript types

```ts
import type { PTSign, PTCategory } from '@road-signs/pt';

type PTCategory =
  | 'information'
  | 'mandatory'
  | 'priority'
  | 'prohibitory'
  | 'warning';

interface PTSign {
  id: string;          // e.g. "a1-danger-bend-right"
  code: string;        // e.g. "A1"
  name: string;
  description: string;
  category: PTCategory;
  svg: string;         // inline SVG string
  assets: {
    svg: string;
    png: Record<number, string>;
    jpg: Record<number, string>;
    webp: Record<number, string>;
  };
}
```
