The `@road-signs/tv` package contains Tuvalu road signs.

## Installation

```sh
npm install @road-signs/tv
```

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/tv';

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("example-sign");

    // By sign code
    const s = getSignByCode("1.1");

    // All warning signs
    const filtered = getSignsByCategory('warning');
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/tv';
    import { RoadSign } from '@road-signs/react';

    const warnings = getSignsByCategory('warning');

    export default function WarningGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {warnings.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
</Tabs>

import type { TVSign, TVCategory } from '@road-signs/tv';
