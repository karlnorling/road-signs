The `@road-signs/vu` package contains Vanuatu road signs.

## Installation

```sh
npm install @road-signs/vu
```

{/* USAGE:START */}

## Usage

<Tabs>
  <TabItem label="TypeScript">
    ```ts
    import { signs, getSign, getSignByCode, getSignsByCategory } from '@road-signs/vu';

    // Example below uses the real VU sign "STOP" — adjust to taste.

    // All signs
    signs.length;

    // By slug ID
    const sign = getSign("stop-stop");

    // By sign code
    const s = getSignByCode("STOP");

    // All priority signs
    const filtered = getSignsByCategory("priority");
    ```
  </TabItem>
  <TabItem label="React">
    ```tsx
    import { getSignsByCategory } from '@road-signs/vu';
    import { RoadSign } from '@road-signs/react';

    const prioritySigns = getSignsByCategory("priority");

    export default function VUGrid() {
      return (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
          {prioritySigns.map((sign) => (
            <RoadSign key={sign.id} sign={sign} size={64} />
          ))}
        </div>
      );
    }
    ```
  </TabItem>
  <TabItem label="Vue 3">
    ```vue
    <script setup>
    import { getSignsByCategory } from '@road-signs/vu';
    import { RoadSign } from '@road-signs/vue';

    const prioritySigns = getSignsByCategory("priority");
    </script>

    <template>
      <div style="display:flex;gap:8px;flex-wrap:wrap">
        <RoadSign v-for="sign in prioritySigns" :key="sign.id" :sign="sign" :size="64" />
      </div>
    </template>
    ```
  </TabItem>
</Tabs>
